// ---------- noise ----------
function hash(x, z) { const s = Math.sin(x * 127.1 + z * 311.7) * 43758.5453; return s - Math.floor(s) }
function vnoise(x, z) {
  const xi = Math.floor(x), zi = Math.floor(z), xf = x - xi, zf = z - zi
  const u = xf * xf * (3 - 2 * xf), v = zf * zf * (3 - 2 * zf)
  const a = hash(xi, zi), b = hash(xi + 1, zi), c = hash(xi, zi + 1), d = hash(xi + 1, zi + 1)
  return a + (b - a) * u + (c - a) * v + (a - b - c + d) * u * v
}
const fbm = (x, z) => { let s = 0, a = 1, f = 1; for (let i = 0; i < 4; i++) { s += a * vnoise(x * f, z * f); a *= 0.5; f *= 2.03 } return s / 1.875 }
const smooth = (e0, e1, x) => { const t = Math.min(1, Math.max(0, (x - e0) / (e1 - e0))); return t * t * (3 - 2 * t) }

export const pathX = z => Math.sin(z * 0.035) * 7
const baseH = z => 30 - z * 0.12 + (z < -60 ? (-60 - z) * 0.25 : 0)
const ZONES = [
  { p: PLACES.temple, r: 15, d: 0 }, { p: PLACES.rock, r: 4, d: 2.5 }, { p: PLACES.village, r: 24, d: 0 },
  { p: PLACES.gate, r: 9, d: 0 }, { p: PLACES.thennur, r: 16, d: 0 }, { p: PLACES.fortress, r: 22, d: 3 }, { p: PLACES.throne, r: 9, d: 4 },
]
for (const z of ZONES) z.h = baseH(z.p.z) + z.d

export function heightAt(x, z) {
  let h = baseH(z)
  const ax = Math.abs(x - pathX(z))
  h += Math.min(70, Math.max(0, ax - 22) ** 1.6 * 0.09)                       // valley walls
  h += (fbm(x * 0.05, z * 0.05) - 0.5) * 7 * smooth(4, 14, ax)  // rough terrain away from path
  h += (fbm(x * 0.2, z * 0.2) - 0.5) * 0.6
  for (const zn of ZONES) {
    const d = Math.hypot(x - zn.p.x, z - zn.p.z)
    const w = 1 - smooth(zn.r, zn.r + 10, d)
    h = h + (zn.h - h) * w
  }
  return h
}
for (const k in PLACES) PLACES[k].y = heightAt(PLACES[k].x, PLACES[k].z)

  // ---------------- sky & light ----------------
  buildSky() {
    this.skyU = { top: { value: new THREE.Color() }, bottom: { value: new THREE.Color() }, sunDir: { value: new THREE.Vector3(0, 1, 0) }, sunCol: { value: new THREE.Color() } }
    const sky = new THREE.Mesh(new THREE.SphereGeometry(900, 32, 16), new THREE.ShaderMaterial({
      side: THREE.BackSide, depthWrite: false, uniforms: this.skyU,
      vertexShader: `varying vec3 vP; void main(){ vP = normalize(position); gl_Position = projectionMatrix*modelViewMatrix*vec4(position,1.); }`,
      fragmentShader: `uniform vec3 top,bottom,sunCol,sunDir; varying vec3 vP;
        void main(){ float h = clamp(vP.y*1.6+0.15,0.,1.); vec3 c = mix(bottom, top, pow(h,0.8));
          float s = max(dot(vP, normalize(sunDir)),0.); c += sunCol*(pow(s,600.)*2.5 + pow(s,8.)*0.35);
          gl_FragColor = vec4(c,1.); }`,
    }))
    sky.frustumCulled = false
    this.sky = sky; this.scene.add(sky)
    this.scene.fog = new THREE.FogExp2(0xffffff, 0.009)
  }
  buildLights() {
    this.hemi = new THREE.HemisphereLight(0xffffff, 0x334422, 0.6); this.scene.add(this.hemi)

    const c = sun.shadow.camera; c.left = -40; c.right = 40; c.top = 40; c.bottom = -40; c.near = 1; c.far = 220
    sun.shadow.bias = -0.0005; sun.shadow.normalBias = 0.04
    this.sun = sun; this.scene.add(sun); this.scene.add(sun.target)
    this.sunOffset = new THREE.Vector3(50, 80, 30)
  }

  // ---------------- terrain ----------------
  buildTerrain() {
    const W = 280, D = 440, sx = 210, sz = 330
    const g = new THREE.PlaneGeometry(W, D, sx, sz); g.rotateX(-Math.PI / 2); g.translate(0, 0, 40)
    const pos = g.attributes.position, col = [], c = new THREE.Color()
    const grass = new THREE.Color(0x5f8a3a), grass2 = new THREE.Color(0x3f6a2a), dirt = new THREE.Color(0x8a6a44), rock = new THREE.Color(0x6e6a64), snow = new THREE.Color(0xeef2f6)
    for (let i = 0; i < pos.count; i++) {
      const x = pos.getX(i), z = pos.getZ(i), h = heightAt(x, z); pos.setY(i, h)
    }
    g.computeVertexNormals()
    const nrm = g.attributes.normal
    for (let i = 0; i < pos.count; i++) {
      const x = pos.getX(i), z = pos.getZ(i), h = pos.getY(i), ny = nrm.getY(i)
      c.copy(grass).lerp(grass2, fbm(x * 0.1, z * 0.1))
      const onPath = 1 - smooth(1.6, 3.2, Math.abs(x - pathX(z)))
      c.lerp(dirt, onPath * 0.85)
      for (const zn of ZONES) if (Math.hypot(x - zn.p.x, z - zn.p.z) < zn.r * 0.8) c.lerp(dirt, 0.35)
      c.lerp(rock, smooth(0.86, 0.7, ny))
      c.lerp(snow, smooth(62, 76, h))
      col.push(c.r, c.g, c.b)
    }
    g.setAttribute('color', new THREE.Float32BufferAttribute(col, 3))
    const m = new THREE.Mesh(g, new THREE.MeshStandardMaterial({ vertexColors: true, flatShading: true, roughness: 0.95 }))
    m.receiveShadow = true; this.terrain = m; this.scene.add(m)
    // distant mountain ring
    const far = new THREE.Group()
    for (let i = 0; i < 26; i++) {
      const a = i / 26 * Math.PI * 2, r = 330 + Math.random() * 60, h = 90 + Math.random() * 120
      const cone = new THREE.Mesh(new THREE.ConeGeometry(70 + Math.random() * 50, h, 5), M(0x5a6680))
      cone.position.set(Math.cos(a) * r, h / 2 - 20, Math.sin(a) * r + 40); far.add(cone)
      const cap = new THREE.Mesh(new THREE.ConeGeometry(cone.geometry.parameters.radius * 0.3, h * 0.3, 5), M(0xf0f4f8)); cap.position.copy(cone.position); cap.position.y += h * 0.35; far.add(cap)
    }
    this.scene.add(far)
  }

  // ---------------- flora ----------------
  buildFlora() {
    const ok = (x, z, pad = 4) => Math.abs(x - pathX(z)) > pad && ZONES.every(zn => Math.hypot(x - zn.p.x, z - zn.p.z) > zn.r + 2)
    // pines
    const trunkG = new THREE.CylinderGeometry(0.15, 0.25, 1.6, 5); trunkG.translate(0, 0.8, 0)
    const leafG = new THREE.ConeGeometry(1.4, 3.6, 6); leafG.translate(0, 3.2, 0)
    const N = 700, trunks = new THREE.InstancedMesh(trunkG, M(0x4a3020), N), leaves = new THREE.InstancedMesh(leafG, M(0x2f5a32), N)
    const d = new THREE.Object3D(); let n = 0
    for (let i = 0; i < 4000 && n < N; i++) {
      const x = (Math.random() - 0.5) * 120, z = -100 + Math.random() * 285
      if (!ok(x, z, 6) || Math.abs(x) < 9 || Math.hypot(x - PLACES.rock.x, z - PLACES.rock.z) < 18) continue
      const h = heightAt(x, z); if (h > 50) continue
      d.position.set(x, h - 0.2, z); d.rotation.y = Math.random() * 6; d.scale.setScalar(0.7 + Math.random() * 0.9); d.updateMatrix()
      trunks.setMatrixAt(n, d.matrix); leaves.setMatrixAt(n, d.matrix)
      leaves.setColorAt(n, new THREE.Color().setHSL(0.3 + Math.random() * 0.05, 0.35, 0.22 + Math.random() * 0.08)); n++
    }
    trunks.count = leaves.count = n; leaves.castShadow = trunks.castShadow = true
    this.scene.add(trunks, leaves)
    // Kurinji shrubs — budded until the bloom
    const FN = 4500
    const shrubG = new THREE.IcosahedronGeometry(0.35, 0)
    const flowerG = new THREE.IcosahedronGeometry(0.16, 0)
    this.shrubs = new THREE.InstancedMesh(shrubG, M(0x3a5a30), FN)
    this.flowers = new THREE.InstancedMesh(flowerG, new THREE.MeshStandardMaterial({ color: 0x8a7cf0, emissive: 0x3a2a9a, emissiveIntensity: 0.4, roughness: 0.6, flatShading: true }), FN)
    this.flowerData = []
    n = 0
    for (let i = 0; i < 20000 && n < FN; i++) {
      const x = (Math.random() - 0.5) * 110, z = -98 + Math.random() * 200
      if (!ok(x, z, 2.2)) continue
      const h = heightAt(x, z); if (h > 64) continue
      if (fbm(x * 0.06 + 10, z * 0.06) < 0.42) continue
      d.position.set(x, h + 0.1, z); d.rotation.set(0, Math.random() * 6, 0); d.scale.setScalar(0.7 + Math.random() * 0.8); d.updateMatrix()
      this.shrubs.setMatrixAt(n, d.matrix)
      this.flowerData.push({ x, y: h + 0.35, z, s: d.scale.x, dist: Math.hypot(x - PLACES.rock.x, z - PLACES.rock.z) })
      n++
    }
    this.shrubs.count = this.flowers.count = n
    this.bloomT = 0; this.setBloom(0)
    this.scene.add(this.shrubs, this.flowers)
    // grass tufts
    const tG = new THREE.ConeGeometry(0.08, 0.5, 3); tG.translate(0, 0.25, 0)
    const tufts = new THREE.InstancedMesh(tG, M(0x7a9a44), 6000); n = 0
    for (let i = 0; i < 9000 && n < 6000; i++) {
      const x = (Math.random() - 0.5) * 100, z = -98 + Math.random() * 280; if (Math.abs(x - pathX(z)) < 2) continue
      const h = heightAt(x, z); if (h > 48) continue
      d.position.set(x, h, z); d.rotation.set((Math.random() - .5) * .4, 0, (Math.random() - .5) * .4); d.scale.setScalar(0.6 + Math.random()); d.updateMatrix(); tufts.setMatrixAt(n++, d.matrix)
    }
    tufts.count = n; this.scene.add(tufts)
  }
  /** b: 0..1 — bloom spreads outward from the meditation rock */
  setBloom(b) {
    this.bloomT = b
    const d = new THREE.Object3D()
    const maxD = 190
    this.flowerData.forEach((f, i) => {
      const local = Math.min(1, Math.max(0, (b * maxD * 1.25 - f.dist) / 18))
      const s = 0.25 + local * 1.15
      d.position.set(f.x, f.y + local * 0.15, f.z); d.scale.setScalar(f.s * s); d.rotation.set(0, i, 0); d.updateMatrix()
      this.flowers.setMatrixAt(i, d.matrix)
    })
    this.flowers.instanceMatrix.needsUpdate = true
    this.flowers.material.emissiveIntensity = 0.3 + b * 0.9
  }

    // fire embers & flames
    const N = 900, geo = new THREE.BufferGeometry(), pos = new Float32Array(N * 3), seed = new Float32Array(N)

    const BN = 400, bg = new THREE.BufferGeometry()

  dotTexture() {
    const c = document.createElement('canvas'); c.width = c.height = 64
    const x = c.getContext('2d'), g = x.createRadialGradient(32, 32, 0, 32, 32, 32)
    g.addColorStop(0, 'rgba(255,255,255,1)'); g.addColorStop(0.3, 'rgba(255,255,255,.6)'); g.addColorStop(1, 'rgba(255,255,255,0)')
    x.fillStyle = g; x.fillRect(0, 0, 64, 64); return new THREE.CanvasTexture(c)
  }
  spawnBurst(p, n = 20, color, speed = 6) {
    const B = this.burst
    if (color) this.burstPts.material.color.setHex(color)
    for (let k = 0; k < n; k++) {
      const i = B.i = (B.i + 1) % B.N
      B.pos[i * 3] = p.x; B.pos[i * 3 + 1] = p.y; B.pos[i * 3 + 2] = p.z
      const a = Math.random() * 6.28, u = Math.random() * 2 - 1, s = speed * (0.4 + Math.random() * 0.6)
      B.vel[i * 3] = Math.cos(a) * Math.sqrt(1 - u * u) * s; B.vel[i * 3 + 1] = Math.abs(u) * s * 0.8 + 1; B.vel[i * 3 + 2] = Math.sin(a) * Math.sqrt(1 - u * u) * s
      B.life[i] = 0.5 + Math.random() * 0.4
    }
  }
  setFires(on) { this.firesOn = on }

  update(dt, t, focus) {
    for (const f of this.anim) f(dt, t)
    // sun follows player for crisp shadows
    this.sun.position.copy(focus).add(this.sunOffset); this.sun.target.position.copy(focus)
    this.sky.position.copy(focus)
    // fire
    const fp = this.firePts.geometry.attributes.position
    const N = this.fireSeed.length, fires = this.fires
    for (let i = 0; i < N; i++) {
      const s = this.fireSeed[i], src = fires[i % fires.length]
      const life = (t * (0.5 + s * 0.6) + s * 7) % 1
      if (!this.firesOn || !src) { fp.setXYZ(i, 0, -999, 0); continue }
      fp.setXYZ(i, src.x + Math.sin(s * 50 + t) * (1.3 - life) * 1.2, src.y - 1 + life * 4.5, src.z + Math.cos(s * 31 + t) * (1.3 - life) * 1.2)
    }
    fp.needsUpdate = true
    // petals drift around focus
    const pp = this.petalPts.geometry.attributes.position
    if (this.petalPts.material.opacity > 0.01) for (let i = 0; i < pp.count; i++) {
      let y = pp.getY(i) - dt * 0.6; if (y < 0) y += 20
      pp.setXYZ(i, pp.getX(i) + Math.sin(t + i) * dt * 0.5, y, pp.getZ(i) + Math.cos(t * 0.7 + i) * dt * 0.4)
    }
    pp.needsUpdate = true; this.petalPts.position.set(focus.x, heightAt(focus.x, focus.z) - 2, focus.z)
    if (this.fireflies.visible) {
      const a = this.fireflies.geometry.attributes.position
      for (let i = 0; i < a.count; i++) a.setXYZ(i, this.ffBase[i * 3] + Math.sin(t * 0.7 + i) * 0.8, this.ffBase[i * 3 + 1] + Math.sin(t * 1.3 + i * 2) * 0.4, this.ffBase[i * 3 + 2] + Math.cos(t * 0.5 + i) * 0.8)
      a.needsUpdate = true
    }
    const B = this.burst
    for (let i = 0; i < B.N; i++) {
      if (B.life[i] <= 0) { B.pos[i * 3 + 1] = -999; continue }
      B.life[i] -= dt; B.vel[i * 3 + 1] -= 12 * dt
      B.pos[i * 3] += B.vel[i * 3] * dt; B.pos[i * 3 + 1] += B.vel[i * 3 + 1] * dt; B.pos[i * 3 + 2] += B.vel[i * 3 + 2] * dt
    }
    this.burstPts.geometry.attributes.position.needsUpdate = true
  }
